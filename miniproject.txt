  console.log(`Grade: ${grade}`);
  console.log(`Status: ${status}`);
}

// ---------- Capstone: Interactive Product Inventory & Analytics ----------
console.log("\n\n=== CAPSTONE ===");
console.log("--- Initializing Inventory System ---");

const inventory = [
  { name: "Laptop", price: 800 },
  { name: "Mouse", price: 25 },
];

// 1. Input with validation (loop + break/continue)
const MAX_ATTEMPTS = 3;
let added = false;
for (let attempt = 1; attempt <= MAX_ATTEMPTS; attempt++) {
  const rawName = ask("Enter product name:", "Monitor");
  const rawPrice = ask("Enter product price:", 150);

  if (rawName === null || rawPrice === null) break; // user cancelled
  const name = String(rawName).trim();
  const price = Number(rawPrice);

  if (name === "" || Number.isNaN(price) || price <= 0) {
    console.log(
      `Invalid input (attempt ${attempt}/${MAX_ATTEMPTS}). Try again.`,
    );
    continue;
  }
  inventory.push({ name, price });
  console.log(`[Prompt executed: User adds "${name}" at "$${price}"]`);
  added = true;
  break;
}
if (!added) console.log("No new product added.");

// 2. Categorize with if/else + switch
const PREMIUM_MIN = 500;
const BUDGET_MAX = 50;

function categorize(price) {
  let tier;
  if (price >= PREMIUM_MIN) tier = "premium";
  else if (price < BUDGET_MAX) tier = "budget";
  else tier = "standard";

  switch (tier) {
    case "premium":
      return "Premium";
    case "budget":
      return "Budget";
    default:
      return "Standard";
  }
}

// 3. Report with forEach
console.log("\n--- Processing Inventory Roster (forEach) ---");
inventory.forEach((item) => {
  console.log(
    `* Item: ${item.name} ($${item.price}) -> Category: ${categorize(item.price)}`,
  );
});

// 4. Functions & scope: stats, tax/discount
function totalInventoryValue(items) {
  let sum = 0; // local scope
  for (const item of items) sum += item.price;
  return sum;
}
const applyAdjustments = (price, taxRate = 0, discountRate = 0) =>
  price * (1 - discountRate) * (1 + taxRate);

// 5. filter + map
const affordable = inventory
  .filter((item) => item.price < 200)
  .map((item) => item.name);
const report = inventory.map(
  (item) => `${item.name.toUpperCase()}: $${item.price}`,
);

console.log("\n--- Financial Analytics ---");
console.log(`Total Inventory Value: $${totalInventoryValue(inventory)}`);
console.log(
  `Filtered Affordable Items (Under $200): ${JSON.stringify(affordable).replace(/,/g, ", ")}`,
);
console.log(
  `Formatted Report List: ${JSON.stringify(report).replace(/,/g, ", ")}`,
);

// Bonus: tax/discount demo
const withTax = applyAdjustments(totalInventoryValue(inventory), 0.18, 0.1);
console.log(
  `(Bonus) Total after 10% discount + 18% tax: $${withTax.toFixed(2)}`,
);
